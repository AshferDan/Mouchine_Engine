# ======================= Project Configuration =========================
cmake_minimum_required(VERSION 2.8...3.13)
# Set the CMake policy version (this enables all policies up to CMake 3.5)
cmake_policy(VERSION 3.5)
project(GlWithCMake)

# Set the directory where custom CMake modules are stored
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")

# Include project-wide utility and dependency modules
include(ImportDependencies)
importDependencies()

# Global compile flags and macros
add_definitions(-DGLEW_STATIC)
add_definitions(-DUSE_CSD3151_AUTOMATION=0) # Used for instructor's automation

# List of external libraries
set(ALL_LIBS
  PRIVATE glfw
  PRIVATE libglew_static
  PRIVATE imgui
)

# Include path for GLM (optional if not using FetchContent)
include_directories(${CMAKE_CURRENT_LIST_DIR}/lib/glm)

# ======================= Graphics Directory =========================
set(Graphics_DIR ./Graphics)
file(GLOB_RECURSE Graphics_SRC 
   "${Graphics_DIR}/*.cpp"
)
file(GLOB_RECURSE Graphics_HEAD 
   "${Graphics_DIR}/*.h"
   "${Graphics_DIR}/*.hpp"
)

# ======================= Transform Directory ========================
set(Transform_DIR ./Transform)
file(GLOB_RECURSE Transform_SRC 
   "${Transform_DIR}/*.cpp"
)
file(GLOB_RECURSE Transform_HEAD 
   "${Transform_DIR}/*.h"
   "${Transform_DIR}/*.hpp"
)

# ======================= Main Directory =============================
set(Main_DIR ./main)
file(GLOB_RECURSE Main_SRC 
   "${Main_DIR}/*.cpp"
)
file(GLOB_RECURSE Main_HEAD 
   "${Main_DIR}/*.h"
   "${Main_DIR}/*.hpp"
)

# ======================= Source Groups for Visual Studio ============
# Create filters in Visual Studio Solution Explorer
source_group("Graphics\\Header Files" FILES ${Graphics_HEAD})
source_group("Graphics\\Source Files" FILES ${Graphics_SRC})

source_group("Transform\\Header Files" FILES ${Transform_HEAD})
source_group("Transform\\Source Files" FILES ${Transform_SRC})

source_group("Main\\Header Files" FILES ${Main_HEAD})
source_group("Main\\Source Files" FILES ${Main_SRC})

# ======================= Create Executable ==========================
add_executable(${CMAKE_PROJECT_NAME}
    # Graphics files
    ${Graphics_HEAD}
    ${Graphics_SRC}
    # Transform files
    ${Transform_HEAD}
    ${Transform_SRC}
    # Main files
    ${Main_HEAD}
    ${Main_SRC}
)

# ======================= Link Libraries =============================
target_link_libraries(${CMAKE_PROJECT_NAME}
    ${ALL_LIBS}
)

# ======================= Set C++ Standard ===========================
set_property(TARGET ${CMAKE_PROJECT_NAME} PROPERTY CXX_STANDARD 20)

# ======================= Compiler Warnings =========================
if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE -Wall ${DisableWarnings})
elseif (MSVC)
    target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE /W3 /WX-)
endif()

# ======================= Include Directories ========================
target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE
    ${Graphics_DIR}
    ${Transform_DIR}
    ${Main_DIR}
)
